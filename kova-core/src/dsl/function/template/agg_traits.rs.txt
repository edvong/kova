impl<'a, T, DB: Database> IntoFunction<'a, DB> for {NAME}<Column<'a, T, DB>> {
    #[inline(always)]
    fn into_func(self) -> Function<'a, DB> {
        self.into_aggregate_with(Quantifier::All)
    }
}

impl<'a, T, DB: Database> IntoAggregate<'a, DB> for {NAME}<Column<'a, T, DB>> {
    #[inline(always)]
    fn into_aggregate_with(self, quantifier: Quantifier) -> Function<'a, DB> {
        Function::from_std(StandardFunction::{NAME}(self.0.into_identifier().meta, quantifier))
    }
}

impl<'a, T, DB: Database> {NAME}<Column<'a, T, DB>> {
    pub fn distinct(self) -> Function<'a, DB> {
        self.into_aggregate_distinct()
    }

    pub fn over<X>(self) -> Window<'a, DB, X> {
        Window::new(self.into_func())
    }
}
